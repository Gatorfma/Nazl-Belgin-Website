# Multilingual Site (Turkish and Arabic) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publish the portfolio at `/`, `/tr/`, and `/ar/` as three separately indexable pages generated from `index.html`, with Arabic rendering right-to-left.

**Architecture:** `index.html` stays the hand-edited English site and doubles as the translation source. Elements carry `data-i18n` / `data-i18n-attr` markers. A Node build script reads `i18n/tr.json` and `i18n/ar.json`, substitutes marked content, rewrites the document head for each language, and writes `tr/index.html` and `ar/index.html`, which are committed. Supabase `series` and `medium` values translate through a client-side dictionary, so no schema change.

**Tech Stack:** Vanilla HTML/CSS/ES5-style JS, UMD modules, `node --test` with `node:assert/strict`, no build dependencies beyond Node's standard library.

**Spec:** `docs/superpowers/specs/2026-10-05-multilingual-tr-ar-design.md`

## Global Constraints

- Three languages only: `en` at `/`, `tr` at `/tr/`, `ar` at `/ar/`.
- `index.html` is never generated. Only `tr/index.html` and `ar/index.html` are build output, and each begins with `<!-- GENERATED by tools/build-i18n.js — edit index.html instead -->`.
- A missing translation key is a build failure (non-zero exit), never a silent English fallback.
- No database schema change, and no edits to `js/studio.js`. The studio UI stays English.
- No automatic language redirect by browser language or IP.
- Artwork titles are never translated. `Untitled` is UI text and is translated.
- JS modules follow the existing UMD wrapper so `node --test` can `require()` them and the browser gets a global.
- Tests run as `node --test tools/<name>.test.js`; assertions use `node:assert/strict`.
- CSS changes must be behaviour-preserving in LTR: English and Turkish layout stays visually identical to today.
- `og:locale` values are exactly `en_US`, `tr_TR`, `ar_AR`. `hreflang` values are exactly `en`, `tr`, `ar`, `x-default`.

## Review Focus

- **Missing key in one language only.** A key present in `tr.json` but absent from `ar.json` must fail the build naming the key and the language, not emit an English-looking Arabic page. Test in Task 5.
- **Translated text containing HTML-special characters.** Turkish and Arabic strings legitimately contain `&`, `<`, `"`, and apostrophes. Substituted text must be escaped so it renders as characters and cannot inject markup. Test in Task 2.
- **Bidirectional text with Latin runs and digits.** Arabic strings mixing Latin words and numbers (`2025 · 4:12`, `Historia Art Studio`) reorder visually under RTL. Isolation must be applied so they read correctly. Test in Task 9.
- **A series value absent from the dictionary.** The studio can add a series after launch. Lookup must fall back to the stored English value, never render `undefined` or empty, and filtering must keep working. Test in Task 11.
- **`index.html` edited without rebuilding.** `tr/` and `ar/` silently drift from the English source. A drift check must detect it. Test in Task 12.

```js
(function (root, factory) {
  'use strict';

  var api = factory();
  if (typeof module === 'object' && module.exports) {
    module.exports = api;
    return;
  }
  root.NBI18nTransform = api;
})(typeof window !== 'undefined' ? window : globalThis, function () {
  'use strict';

  var TEXT_MARKER = /\sdata-i18n="([^"]+)"/g;
  var ATTR_MARKER = /\sdata-i18n-attr="([^"]+)"/g;

  function parseAttrSpec(spec) {
    return String(spec).split(',').map(function (part) {
      var bits = part.split(':');
      return {
        attribute: String(bits[0] || '').trim(),
        key: String(bits[1] || '').trim()
      };
    }).filter(function (entry) {
      return entry.attribute && entry.key;
    });
  }

  function unique(list) {
    var seen = Object.create(null);
    var out = [];
    list.forEach(function (value) {
      if (seen[value]) return;
      seen[value] = true;
      out.push(value);
    });
    return out.sort();
  }

  function collectKeys(html) {
    var text = [];
    var attr = [];
    var match;

    TEXT_MARKER.lastIndex = 0;
    while ((match = TEXT_MARKER.exec(html)) !== null) text.push(match[1]);

    ATTR_MARKER.lastIndex = 0;
    while ((match = ATTR_MARKER.exec(html)) !== null) {
      parseAttrSpec(match[1]).forEach(function (entry) { attr.push(entry.key); });
    }

    return { text: unique(text), attr: unique(attr) };
  }

  return {
    collectKeys: collectKeys,
    parseAttrSpec: parseAttrSpec
  };
});
```

```js
  function escapeText(value) {
    return String(value)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;');
  }

  function escapeAttr(value) {
    return escapeText(value)
      .replace(/"/g, '&quot;')
      .replace(/'/g, '&#39;');
  }

  function applyTranslations(html, dict) {
    var out = html.replace(
      /(<([a-zA-Z0-9-]+)\b[^>]*\sdata-i18n="([^"]+)"[^>]*>)([\s\S]*?)(<\/\2>)/g,
      function (whole, open, tag, key, body, close) {
        if (!Object.prototype.hasOwnProperty.call(dict, key)) return whole;
        return open + escapeText(dict[key]) + close;
      }
    );

    out = out.replace(
      /<([a-zA-Z0-9-]+)\b([^>]*\sdata-i18n-attr="([^"]+)"[^>]*)>/g,
      function (whole, tag, attrs, spec) {
        var rewritten = attrs;
        parseAttrSpec(spec).forEach(function (entry) {
          if (!Object.prototype.hasOwnProperty.call(dict, entry.key)) return;
          var pattern = new RegExp('(\\s' + entry.attribute + '=")[^"]*(")');
          rewritten = rewritten.replace(
            pattern,
            '$1' + escapeAttr(dict[entry.key]).replace(/\$/g, '$$$$') + '$2'
          );
        });
        return '<' + tag + rewritten + '>';
      }
    );

    return out;
  }
```

```js
  var SITE = 'https://nazlibelgin.com/';
  var LANGS = ['en', 'tr', 'ar'];
  var LOCALES = { en: 'en_US', tr: 'tr_TR', ar: 'ar_AR' };

  function urlFor(lang) {
    return lang === 'en' ? SITE : SITE + lang + '/';
  }

  function hreflangBlock() {
    var links = LANGS.map(function (lang) {
      return '<link rel="alternate" hreflang="' + lang + '" href="' + urlFor(lang) + '">';
    });
    links.push('<link rel="alternate" hreflang="x-default" href="' + urlFor('en') + '">');
    return links.join('\n');
  }

  function rewriteHead(html, lang) {
    var openTag = lang === 'ar'
      ? '<html lang="ar" dir="rtl">'
      : '<html lang="' + lang + '">';

    var out = html.replace(/<html[^>]*>/, openTag);

    out = out.replace(
      /(<link rel="canonical" href=")[^"]*(")/,
      '$1' + urlFor(lang) + '$2'
    );
    out = out.replace(
      /(<meta property="og:url" content=")[^"]*(")/,
      '$1' + urlFor(lang) + '$2'
    );
    out = out.replace(
      /(<meta property="og:locale" content=")[^"]*(")/,
      '$1' + LOCALES[lang] + '$2'
    );

    return out.replace(
      /(<link rel="canonical"[^>]*>)/,
      '$1\n' + hreflangBlock()
    );
  }
```

In `tools/i18n-transform.js`:

```js
  function isSourceKey(key) {
    return /\._source$/.test(key);
  }

  function validateKeys(html, dict) {
    var found = collectKeys(html);
    var used = found.text.concat(found.attr).filter(function (key, index, list) {
      return list.indexOf(key) === index;
    }).sort();

    var missing = used.filter(function (key) {
      return !Object.prototype.hasOwnProperty.call(dict, key);
    });

    var orphaned = Object.keys(dict).filter(function (key) {
      return !isSourceKey(key) && used.indexOf(key) === -1;
    }).sort();

    return { missing: missing, orphaned: orphaned };
  }
```

Create `tools/build-i18n.js`:

```js
#!/usr/bin/env node
'use strict';

var fs = require('node:fs');
var path = require('node:path');
var transform = require('./i18n-transform.js');

var ROOT = path.join(__dirname, '..');
var BANNER = '<!-- GENERATED by tools/build-i18n.js — edit index.html instead -->';

function loadDictionary(lang) {
  var file = path.join(ROOT, 'i18n', lang + '.json');
  return JSON.parse(fs.readFileSync(file, 'utf8'));
}

function build() {
  var source = fs.readFileSync(path.join(ROOT, 'index.html'), 'utf8');
  var failed = false;

  transform.LANGS.filter(function (lang) { return lang !== 'en'; })
    .forEach(function (lang) {
      var dict = loadDictionary(lang);
      var report = transform.validateKeys(source, dict);

      if (report.missing.length) {
        failed = true;
        console.error('[' + lang + '] missing ' + report.missing.length + ' key(s):');
        report.missing.forEach(function (key) { console.error('  - ' + key); });
      }
```

```js
  assert.ok(page('ar').length > 0);
```
